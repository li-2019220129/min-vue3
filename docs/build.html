45.1 processText / processComment

function processText(n1, n2, container, anchor) {
  if (n1 == null) {
    n2.el = hostCreateText(n2.children);
    hostInsert(n2.el, container, anchor);
  } else {
    const el = (n2.el = n1.el);
    if (n2.children !== n1.children) hostSetText(el, n2.children);
  }
}

45.2 processFragment：双锚点

function processFragment(n1, n2, container, anchor) {
  const fragmentStartAnchor = (n2.el = hostCreateText(""));
  const fragmentEndAnchor = (n2.anchor = hostCreateText(""));
  if (n1 == null) {
    hostInsert(fragmentStartAnchor, container, anchor);
    hostInsert(fragmentEndAnchor, container, anchor);
    mountChildren(n2.children, container, fragmentEndAnchor);
  } else {
    patchChildren(n1, n2, container, fragmentEndAnchor);
  }
}
// 卸载：unmountFragment 从 startAnchor 沿 nextSibling 删到 endAnchor

45.3 静态节点：免检通道

patch 开头加 25.4 的分支：patchFlag === HOISTED 时把 n2.el = n1.el 直接返回。静态 vnode 的挂载用一次性 hostSetElementText 灌 innerHTML 即可。

⚠️ Fragment 的 anchor 传参陷阱
mountChildren 的 anchor 必须是 fragmentEndAnchor（不是父级 anchor）—— 否则并发多个 Fragment 时子元素会插串行。锚点的意义在这一刻才真正体现。

export const vShow = {
  beforeMount(el, { value }) { el._vod = el.style.display = value ? "" : "none"; },
  updated(el, { value, oldValue }) {
    if (!value === !oldValue) return;
    el.style.display = value ? el._vod || "" : "none";
  },
};
export const vModelText = {
  created(el, { modifiers }, vnode) {
    el.addEventListener(modifiers.lazy ? "change" : "input", () => {
      let v = el.value; if (modifiers.trim) v = v.trim();
      if (modifiers.number) v = looseToNumber(v);
      getModelAssigner(vnode)(v);            // → emit update:modelValue
    });
  },
  beforeUpdate(el, { value, oldValue }) {
    el._assign = getModelAssigner(vnode);
    if (document.activeElement === el && value === oldValue) return;  // 输入中不覆盖
    el.value = value;
  },
};

47.1 生产级调度器（23 章源码直接抄）

const queue = []; let flushIndex = 0, currentFlushId = 0, isFlushing = false, isFlushPending = false;
const pendingPostFlushCbs = []; let currentFlushPromise = null;
const resolvedPromise = Promise.resolve();

export function queueJob(job) {
  if (!queue.includes(job, flushIndex) && (!job.id || job.id > currentFlushId)) queue.push(job);
  queueFlush();
}
export function nextTick(fn) { return (currentFlushPromise || resolvedPromise).then(fn); }

function queueFlush() {
  if (!isFlushing && !isFlushPending) {
    isFlushPending = true;
    currentFlushPromise = resolvedPromise.then(flushJobs);
  }
}
function flushJobs(seen) {
  isFlushPending = isFlushing = true;
  try {
    while (queue.length) {
      queue.sort((a, b) => getId(a) - getId(b));      // 父先子后
      flushPreFlushCbs(seen);                          // beforeUpdate / watch pre
      const job = queue.shift(); job();                // 组件 update → render → patch
      currentFlushId = getId(job); flushIndex++;
    }
    flushPostFlushCbs(seen);                           // mounted / updated / ref 赋值
  } finally {
    isFlushing = false; flushIndex = 0; currentFlushId = 0;
    if (queue.length || pendingPostFlushCbs.length) flushJobs(seen);  // 100 层保护
  }
}
