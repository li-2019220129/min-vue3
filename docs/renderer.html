渲染器：分层的艺术

响应式系统已经毕业，现在换个口味：看 Vue 怎么把「数据」变成「页面上的像素」。 本章有两个宝藏：跨平台的 nodeOps 适配器，和一个被无数框架抄袭的事件缓存技巧 —— invoker。

15.1 nodeOps：把 DOM 装进「接口盒子」

第 1 章说过：runtime-core 不认识 DOM。但它渲染时确实需要「插入节点、删除节点、改文本」这些动作。 怎么办？我不亲自干，我定义标准，谁来谁干。runtime-dom 交上了自己的答卷： 

// 主要是对节点元素的增删改查
export const nodeOps = {
  insert(el, parent, anchor) {
    parent.insertBefore(el, anchor || null);   // 插入（anchor 指定插到谁前面）
  },
  remove(el) {
    const parent = el.parentNode;
    parent && parent.removeChild(el);          // 删除
  },
  createElement: (type) => document.createElement(type),  // 创建元素
  createText: (text) => document.createTextNode(text),    // 创建文本节点
  setText: (node, text) => (node.nodeValue = text),       // 改文本节点的字
  setElementText: (el, text) => (el.textContent = text),  // 改元素的内容
  parentNode: (node) => node.parentNode,
  nextSibling: (node) => node.nextSibling,
};

八个纯函数，全是 DOM API 的薄封装。然后 runtime-dom 把它们和 patchProp 打包，注入给 core：

// runtime-dom：组装渲染选项（接口实现）
export const renderOptions = Object.assign(nodeOps, { patchProp });

export const render = (vnode, container) => {
  createRenderer(renderOptions).render(vnode, container);  // 依赖注入！
};

// runtime-core：拿到一整套「操作能力」，但不知道背后是浏览器
export function createRenderer(renderOptions) {
  const {
    insert: hostInsert,
    remove: hostRemove,
    createElement: hostCreateElement,
    setElementText: hostSetElementText,
    setText: hostSetText,
    parentNode: hostParentNode,
    nextSibling: hostNextSibling,
    patchProp: hostPatchProp,
  } = renderOptions;

  // core 不关心如何渲染
  const render = (vnode, container) => {
    console.log(vnode, container);   // 🚧 项目进度：render 骨架已立，等你来写
  };
  return { render };
}
🧠 这就是「依赖倒置」
高层模块（core）定义抽象接口，低层模块（dom）提供具体实现，通过参数注入。 换一个 runtime-xxx 就能渲染到 canvas、WebGL、原生 App —— 响应式系统一行动不用改。 框架的跨平台能力，从来不是靠兼容，而是靠分层。

15.2 patchProp：属性的分流处理

vnode 的 props 千奇百怪，patchProp 按类型分流到四个处理模块：

function patchProp(el, key, prevValue, nextValue) {
  if (key === "class") {
    return patchClass(el, nextValue);          // class 走 className（比 setAttribute 快）
  } else if (key === "style") {
    return patchStyle(el, prevValue, nextValue);  // style 要做新旧 diff
  } else if (/^on[^a-z]/.test(key)) {          // onClick、oninput → 事件
    return patchEvent(el, key, nextValue);
  } else {
    return patchAttr(el, key, nextValue);      // 其他 → setAttribute
  }
}

四个小模块中最普通的是 attr：

// patchAttr：有值设值，没值删掉
export default function patchAttr(el, key, nextValue) {
  if (nextValue) el.setAttribute(key, nextValue);
  else el.removeAttribute(key);
}

// patchClass：清空用 removeAttribute（注意不是设空字符串）
export default function patchClass(el, value) {
  if (!value) el.removeAttribute("class");
  else el.className = value;
}

// patchStyle：新样式逐条应用；旧样式里新值没有的 → 清空
export default function patchStyle(el, prevValue, nextValue) {
  let style = el.style;
  for (let key in nextValue) style[key] = nextValue[key];
  if (prevValue) {
    for (let key in prevValue) {
      if (nextValue[key] == null) style[key] = '';   // 清掉被删掉的样式（官方用空串，不是 null）
    }
  }
}

15.3 ⭐ patchEvent：invoker，事件缓存的绝招

思考题：每次渲染都会给元素挂新的 onClick 函数。最笨的写法：

el.removeEventListener("click", oldFn);   // 摘掉旧的
el.addEventListener("click", newFn);       // 挂上新的
// ⚠️ 每次渲染都来这么一下，两个昂贵的 DOM 调用

源码的解法妙到毫巅：真正挂到元素上的永远是一个不变的 invoker 壳子， 用户的回调藏在 invoker.value 里随时替换：

function createInvoker(value) {
  const invoker = (e) => invoker.value(e);   // 壳子：点击时才去找此刻的 value
  invoker.value = value;                     // 回调挂在壳子的 value 属性上
  return invoker;
}

export default function patchEvent(el, name, nextValue) {
  const invokers = el._vei || (el._vei = {});     // 元素级缓存表：_vei = {onClick: invoker}
  const eventName = name.slice(2).toLowerCase();
  const existingInvoker = invokers[name];

  if (nextValue && existingInvoker) {
    return (existingInvoker.value = nextValue);   // 情况A：已挂过 → 只换 value！0 次 DOM 调用
  }
  if (nextValue) {
    const invoker = (invokers[name] = createInvoker(nextValue));
    return el.addEventListener(eventName, invoker);  // 情况B：首次 → 挂壳子
  } else if (existingInvoker) {
    el.removeEventListener(eventName, existingInvoker); // 情况C：新值为空 → 摘除
    invokers[name] = undefined;
  }
}
💡 比喻时间：公司前台
addEventListener 像雇一位前台：真正的前台（invoker）永远不换人； 客户来了（点击发生）前台才喊当前负责的同事（invoker.value 回调）。 换负责同事（更新回调）只需换工牌，不用解雇再招聘前台 —— 频繁变化的回调，包一层不变的外壳。React 合成事件、小程序框架里都有同款思想。

15.6 unmount：卸载一条线的完整流程

渲染器三条线（挂载/更新/卸载）里，本章只讲了前两条。卸载虽然短，但顺序同样讲究：

function unmount(vnode) {
  const { shapeFlag, children, component } = vnode;

  // ① 卸载前钩子：指令 beforeUnmount（同步）、组件 beforeUnmount（pre 队列）
  if (vnode.dirs) invokeDirectiveHook(vnode, null, null, 'beforeUnmount');

  // ② KeepAlive 特判（第 24 章）：不真删，挪进缓存容器
  if (shapeFlag & ShapeFlags.COMPONENT_SHOULD_KEEP_ALIVE) {
    vnode.ctx.deactivate(vnode);
    return;
  }

  // ③ 按类型递归卸载
  if (shapeFlag & ShapeFlags.COMPONENT) {
    unmount(component.subTree, null, null);       // 组件：卸载整棵渲染子树
  } else if (shapeFlag & ShapeFlags.FRAGMENT) {
    removeFragment(vnode);                         // Fragment：沿锚点逐个删（第 25 章）
  } else if (shapeFlag & ShapeFlags.TELEPORT) {
    vnode.type.process(vnode, null, null, null);   // Teleport：走自己的分流（第 24 章）
  } else {
    hostRemove(vnode.el);                          // 普通元素：一步 remove
  }

  // ④ 卸载后钩子：unmounted 进 post 队列（此刻 DOM 已不在）
  queuePostRenderEffect(() => {
    if (vnode.dirs) invokeDirectiveHook(vnode, null, null, 'unmounted');
  });
}
🧠 卸载顺序的三个「必须」
① 钩子必须早于 remove：beforeUnmount 里你还要读 DOM、做清理收尾，删了就没得读了； ② 组件必须递归到 subTree：组件 el 只是个壳（第 19 章），真正占 DOM 的是渲染子树， 不递归就漏删；③ unmounted 必须 post 队列：和 mounted 同理，要等整个卸载动作彻底完成， 钩子里看到的世界才是「已卸载」的世界。

15.6 property 还是 attribute？官方 patchProp 的最后一道分流

本项目的 patchProp 四分流（class/style/on/attr）之后还有一个隐藏问题： 走到 attr 分支的键，到底该 setAttribute 还是直接赋值给 DOM Property？ 两者不等价 —— 这道判断在官方叫 shouldSetAsProp： 

function shouldSetAsProp(el, key, value, isSVG) {
  if (isSVG) { ... return false; }              // svg 走 attribute
  if (key === 'form') return false;             // 表单元素的 form 是只读 property
  if (key === 'list' && el.tagName === 'INPUT') return false;
  if (key === 'type' && el.tagName === 'TEXTAREA') return false;
  if (key in el) return true;                   // ⭐ 核心判断：元素上存在同名 property
  ...
}

HTML attribute 是「标签上写死的初始值」，DOM property 是「元素对象上的实时状态」—— 浏览器解析后 attribute 只是个「出厂快照」。Vue 替你选对轨道，就是这套判断的全部意义。 面试问「attribute 和 property 区别」，举 input.value 一例就赢。

15.7 patchElement 的快慢车道：patchFlag 的运行时消费

教学版 patchElement 是「新旧 props 全量对比」。官方编译产物带着 patchFlag（30 章）， 运行时据此直接上快车道 —— 这是「编译反哺运行时」最直白的一段运行时代码： 

const patchElement = (n1, n2, parentComponent) => {
  const el = (n2.el = n1.el);
  const { patchFlag, dynamicChildren, dirs } = n2;
  const oldProps = n1.props || EMPTY_OBJ, newProps = n2.props || EMPTY_OBJ;

  // 官方顺序：⭐ 先比 props（快/慢车道），最后才 patchChildren —— 与 mountElement 相反
  if (patchFlag > 0) {
    if (patchFlag & PatchFlags.FULL_PROPS) {
      patchFullProps(el, oldProps, newProps);        // 慢车道：全量 diff（教学版逻辑）
    } else {
      if (patchFlag & PatchFlags.CLASS) hostPatchClass(el, newProps.class);  // class 专属快道
      if (patchFlag & PatchFlags.STYLE) hostPatchStyle(...);                 // style 专属快道
      if (patchFlag & PatchFlags.PROPS) {
        // ⭐ 快车道：只比编译期登记过的 dynamicProps 数组
        for (let i = 0; i < n2.dynamicProps!.length; i++) {
          const key = n2.dynamicProps![i];
          const next = newProps[key], prev = oldProps[key];
          if (next !== prev) hostPatchProp(el, key, prev, next);
        }
      }
    }
  }

  patchChildren(n1, n2, el, null);       // 子节点最后处理（TEXT / ARRAY 按 shapeFlag 分流）
  ...
};
车道何时走成本
快车道模板编译产物（带 dynamicProps 数组）只比 N 个动态键，通常 0~3 个
慢车道手写渲染函数 / FULL_PROPS 标记新旧 props 全量键遍历
🧠 和 30 章合上盖子
25 章讲「编译器怎么打 patchFlag」，本节补上「运行时怎么按 flag 挑车道」—— 优化闭环到此合龙：编译期多存一个 dynamicProps 数组，运行时每次更新省掉一整轮全量键遍历。 面试被问「patchFlag 到底有什么用」，把快慢车道讲出来就是满分答案。
自测一下：渲染器

1. patchStyle 为什么要两个 for 循环？只循环 nextValue 会怎样？
2. invoker 方案里，事件回调更新时的开销是什么量级？

答案：
1. 第一个循环应用新样式；第二个循环清理「旧样式里有、新样式里没有」的键 —— 缺了它，{color:'red'} 换成 {} 时红色会残留。和新旧依赖清理一个道理：更新 = 加新的 + 删旧的。
2. 一次普通对象属性赋值（invoker.value = fn），O(1)，零 DOM 调用。相比摘除+重挂的两次 DOM API，快得不在一个次元。

h 与 VNode：虚拟的代价与回报

从本章起，我们亲手把 runtime-core 的骨架写满。第一块砖：虚拟节点（VNode）—— 用 JS 对象描述界面，而不是直接操作界面。为什么多此一举？答案藏在一个字里：diff。

16.1 为什么需要 VNode

直接操作 DOM 有什么问题？三个：

  1. DOM 操作昂贵：一次 appendChild 可能触发重排/重绘，代价是 JS 对象操作的百倍量级；
  2. 无法比较：两个真实 DOM 之间没有「新旧可比性」——改了就是改了，覆水难收；
  3. 强绑浏览器：document 只存在于浏览器（还记得第 15 章的跨平台野心吗？）。

JS 对象便宜、可保存、可比较、可跨平台 —— VNode = 界面某一刻的快照。渲染器的工作就是：拍快照 → 新旧快照 diff → 把差异最小化地应用到真实 DOM。

旧 VNode 树 + 新 VNode 树 → diff（纯 JS，超快） → 最小 DOM 更新
昂贵操作（DOM）被「批量 + 最小化」了，便宜操作（比较对象）随便造 —— 这就是虚拟 DOM 的全部经济学
VNode = 渲染快照：对比两张照片，只改不同的地方

16.2 VNode 结构解剖

{
  __v_isVNode: true,      // 身份暗号（IS_REACTIVE 的老朋友了）
  type: "div",            // 标签名 / 组件对象 / Text / Fragment…
  key: null,              // diff 复用的身份证（第 18 章主角）
  props: { class: "card", onClick: fn },
  children: "文本" 或 [vnode, vnode, ...],
  el: null,               // ⭐ 挂载后指向真实 DOM（VNode 与真实世界的桥）
  shapeFlag: ShapeFlags.ELEMENT,   // 类型位标记（下节）
}

三个字段值得划重点：

  • el：挂载时渲染器把真实 DOM 写回 vnode.el —— diff 阶段顺着 el 就能找到真实节点，新旧 vnode 通过它「共用」一个 DOM；
  • key：用户不传就用 null。为什么列表必须传 key？第 18 章你会看到没有 key 时算法直接退化；
  • children 的两种形态：字符串（文本子节点）或数组（子节点列表）—— mount 时分两条路径处理。

16.3 shapeFlag：一个数字背着 32 个布尔值

VNode 可能是元素/文本/组件/Fragment，children 可能是文本/数组/插槽……如果每个可能性都存一个布尔字段，对象会又胖又慢。官方用位运算把所有类型压进一个数字：

const ShapeFlags = {
  ELEMENT: 1,          // 0001  是元素
  FUNCTIONAL: 1 << 1,  // 0010  是函数组件
  STATEFUL: 1 << 2,    // 0100  是有状态组件
  TEXT_CHILDREN: 1 << 3,       // 1000  children 是文本
  ARRAY_CHILDREN: 1 << 4,     // 10000 children 是数组
};

// 标记：一个数字同时记住多个事实
let flag = ShapeFlags.ELEMENT | ShapeFlags.TEXT_CHILDREN;   // 1001 = 9

// 判断：一次 & 运算，比读属性还快
if (flag & ShapeFlags.ELEMENT) { /* 是元素 */ }
if (flag & ShapeFlags.ARRAY_CHILDREN) { /* children 是数组 */ }
💡 位运算为什么快
flag & 8 是一次 CPU 位与指令；而 obj.isElement 是一次属性查找（可能走原型链）。 VNode 创建频率极高（每帧每节点），这种微观优化累积起来就是宏观性能。 Linux 的文件权限（755）、React 的 fiber flags、Vue 的 patchFlags —— 全是同一招。

16.4 h()：创建 VNode 的工厂

直接写对象字面量太啰嗦，Vue 提供 h()（hyperscript，源自 HTML 的发明者超文本思路）：

export function h(type, props, children) {
  // 参数归一化：h("div") / h("div", props) / h("div", [], ...) 都合法
  if (!props) props = {};

  let shapeFlag = typeof type === "string"
    ? ShapeFlags.ELEMENT          // 字符串 → 元素
    : ShapeFlags.STATEFUL;        // 对象 → 组件（第 19 章）

  if (typeof children === "string") {
    shapeFlag |= ShapeFlags.TEXT_CHILDREN;
  } else if (Array.isArray(children)) {
    shapeFlag |= ShapeFlags.ARRAY_CHILDREN;
  }

  return {
    __v_isVNode: true,
    type, props,
    key: props && props.key != null ? props.key : null,
    children,
    el: null,
    shapeFlag,
  };
}

// 使用对照：
// 模板 <div class="card" @click="fn">hi</div> 等价于：
h("div", { class: "card", onClick: fn }, "hi");
// children 是数组的版本：
h("ul", null, [ h("li", { key: "a" }, "苹果"), h("li", { key: "b" }, "香蕉") ]);

16.5 VNode 树 ≠ 组件树

一个容易混淆的点，先种个种子（第 19 章展开）：

VNode 树是「界面结构」的快照，其中某些节点的 type 是组件对象； 组件树是「逻辑单元」的嵌套。一个组件 VNode 展开后是一整棵元素 VNode 子树。
🧠 为什么叫 h？
h = hyperscript，"HyperText Markup Language" 里的 HTML 拆开就是 Hyperscript。 这个命名来自虚拟 DOM 的鼻祖 hyperscript 生态（React.createElement 的思想同源）。 知道来历，你就再也不会把 h 和 hash、height 之类搞混了。

16.6 normalizeChildren：children 的七十二变与一统天下

你写 h('div', null, 'hello')、 h('div', null, [h('span')])、 h(Child, null, { default: () => ... }) —— children 的写法五花八门。官方 normalizeChildren 负责在 createVNode 里把它们归一：

你传的 children归一结果shapeFlag 标记
字符串原样存 children，等 setElementTextTEXT_CHILDREN
数组逐项校验/归一（嵌套数组拍平）ARRAY_CHILDREN
组件 + 单个函数包成 { default: fn }SLOTS_CHILDREN
组件 + 插槽对象直接挂 children（编译器已保证是函数袋）SLOTS_CHILDREN
🧠 为什么组件传函数要写成 () => ...
传 h(Child, null, someVNode) 时，someVNode 在父组件渲染时就创建了； 传 () => someVNode（或对象形式插槽）则把创建推迟到子组件渲染时。 差别在：函数形式能拿到子组件的渲染上下文（作用域插槽的基础，第 22.4 节）， 且父组件更新时可以复用同一个函数引用，减少子组件被误判「插槽变了」的无效更新 —— 编译器给槽对象打的 STABLE 标记（第 22 章）就是给这层优化用的。

16.6 createVNode 的预处理：children 规范化与属性合并

h() 不是简单 new 一个对象 —— 它在创建时就做两类重活。 理解它们，才能看懂模板编译产物里那些「奇怪的参数」： 

function normalizeChildren(vnode, children) {
  if (typeof children === 'string' || typeof children === 'number') {
    type = ShapeFlags.TEXT_CHILDREN;        // ① 文本孩子：setElementText 直写
  } else if (Array.isArray(children)) {
    type = ShapeFlags.ARRAY_CHILDREN;       // ② 数组孩子：走 diff
  } else if (typeof children === 'object') {
    if (children.default) type = ShapeFlags.SLOTS_CHILDREN;  // ③ 槽对象（22.4）
  } else if (children === null) {
    children = null;                        // ④ 清空
  }
}

🧠 为什么在「创建时」做，而不是「patch 时」做？

mountElement：首挂载实战

本章我们把 render 函数从 console.log 写成真正的渲染器： 递归地把 VNode 树翻译成真实 DOM。写完后你会突然明白 —— 原来第 2 章的 effect 和第 17 章的 render，只差一层窗户纸。

17.1 总路线图

render(vnode, container) → patch(null, vnode) → mountElement(vnode, container)
patch 是统一的入口：oldVNode 为 null 就是「挂载」，否则是「更新」（下一章）

17.2 完整实现（可直接抄进 runtime-core/src/index.ts）

export function createRenderer(renderOptions) {
  const {
    insert: hostInsert, remove: hostRemove,
    createElement: hostCreateElement,
    setElementText: hostSetElementText,
    setText: hostSetText,
    parentNode: hostParentNode, nextSibling: hostNextSibling,
    patchProp: hostPatchProp,
  } = renderOptions;

  /* ---------- 挂载一个元素 vnode ---------- */
  function mountElement(vnode, container, anchor) {
    // ① 把 el 回写进 vnode（第 16 章说的那座桥）
    const el = (vnode.el = hostCreateElement(vnode.type));

    // ② 上属性：遍历 props，分流给 patchProp（第 15 章的四分流）
    for (const key in vnode.props) {
      hostPatchProp(el, key, null, vnode.props[key]);
    }

    // ③ 挂 children：文本 / 数组 两条路（shapeFlag 决定）
    if (typeof vnode.children === "string") {
      hostSetElementText(el, vnode.children);
    } else if (Array.isArray(vnode.children)) {
      for (const child of vnode.children) {
        // 数组 children：每个子节点递归 patch(null, child) 挂进来
        patch(null, child, el, null);
      }
    }

    // ④ 最后一步才把 el 插入文档（先在「内存里」组装好，一次进场）
    hostInsert(el, container, anchor);
  }

  /* ---------- 卸载 ---------- */
  function unmount(vnode) {
    hostRemove(vnode.el);
  }

  /* ---------- 统一入口 ---------- */
  function patch(oldVNode, vnode, container, anchor) {
    if (oldVNode === vnode) return;          // 同一个对象引用，啥也不用干
    if (oldVNode && oldVNode.type !== vnode.type) {
      // 类型都变了：旧树整个卸掉，走全新挂载
      unmount(oldVNode);
      oldVNode = null;
    }
    // 统一交给 processElement（预留组件分支的位置，第 19 章）
    processElement(oldVNode, vnode, container, anchor);
  }

  function processElement(oldVNode, vnode, container, anchor) {
    if (oldVNode == null) {
      mountElement(vnode, container, anchor);   // 无旧 → 挂载
    } else {
      patchElement(oldVNode, vnode);            // 有旧 → 更新（下一章实现）
    }
  }

  function render(vnode, container) {
    if (vnode == null) {
      // 传 null → 卸载整棵树（Vue3 卸载组件的官方姿势）
      if (container._vnode) unmount(container._vnode);
    } else {
      patch(container._vnode || null, vnode, container, null);
    }
    container._vnode = vnode;   // container 记住当前树，下次更新 diff 用
  }

  return { render };
}

四个细节，四个心意

  • 「先组装后进场」（④ 在 ③ 后面）：子树全部在游离 DOM 上构建完才 insert，避免布局引擎看到中间状态，减少无谓重排 —— 和装修完再搬家一个道理；
  • container._vnode：渲染器把「上一次的树」记在容器身上。render 两次，自动获得 diff 能力 —— 状态就存在 DOM 节点上，很朴素；
  • anchor 参数一路透传：为 diff 阶段「往中间插节点」预留的位置参数（insert 的第三个参数）；
  • oldVNode === vnode 短路：同一个 vnode 对象直接跳过 —— 第 30 章编译优化的静态提升就靠它白嫖。

17.4 mountElement 的顺序为什么这么讲究

官方 mountElement 的动作顺序是精心排过的，换一步都会出 bug —— 注意官方的顺序和 17.2 教学版恰好相反：先子节点、后属性：

function mountElement(vnode, container, anchor) {
  let el = (vnode.el = hostCreateElement(vnode.type));

  // ① 先挂子节点（文本 or 数组）—— 官方注释原文：
  //    "mount children first, since some props may rely on child content
  //     being already rendered, e.g. <select value>"
  if (shapeFlag & TEXT_CHILDREN) hostSetElementText(el, children);
  else if (shapeFlag & ARRAY_CHILDREN) mountChildren(children, el, null);

  // ② 再打属性
  if (props) { for (const key in props) hostPatchProp(el, key, null, props[key]); }

  // ③ 最后才插入容器 ⭐
  hostInsert(el, container, anchor);

  // ④ 指令 mounted / Transition appear（post 队列）
  if (dirs) queuePostRenderEffect(() => dirs.mounted && ..., anchor);
}

两个反直觉的顺序，各有原因：

  1. 子节点先于属性：某些属性的赋值依赖「子内容已经就位」。官方注释点名的案例是 <select value="x"> —— value 要在 option 挂上之后设置才会生效， 先设会被浏览器重置掉；
  2. 插入容器最后做：整棵子树先在「游离 DOM」里组装完（17.2 的核心技巧）， 一次性 insert —— 避免中间态闪屏，也让子节点的挂载钩子执行时看到的是完整家庭。
💡 SVG 与命名空间
createElement 还要处理 svg / MathML 命名空间： 官方用 getNamespace() 沿 vnode 树继承（<svg> 的子孙全是 svg 命名空间）， 最终调 document.createElementNS。本项目教学版省掉了 —— 你画 SVG 图标库时如果元素不显示，先想想是不是命名空间丢了。

17.4 unmount：mount 的镜像，但多一件必做的事

unmount(vnode) → beforeUnmount 钩子（pre 时机） → 组件线：instance.scope.stop()（第 10 章！）
递归 unmount 所有 children → hostRemove(el)（Fragment 沿锚点删，第 25 章） → unmounted 钩子（post 队列）
顺序必须先深后浅：孩子全部拆完，最后才拆自己 —— 和 mount 正好相反

为什么说「多一件必做的事」？mount 是纯粹的创建，unmount 却要拆干净自己注册过的一切： 渲染 effect（scope.stop）、指令钩子 unmounted、过渡的 leave 缓刑（第 27 章）、 Teleport 子树的目标清理（第 24 章）。 前端内存泄漏排查的第一站永远是「卸载链有没有走完」—— devtools 里能看到 detached DOM 还挂着事件监听，就是 unmount 链断了的铁证。

答案：

2. 卸载 container._vnode 指向的整棵树。对应 unmount() / app.unmount() 的底层。

patch 与 diff：最长递增子序列

渲染器的珠穆朗玛峰：当两组带 key 的子节点都变了， 如何用最少的 DOM 移动让旧列表变成新列表？Vue3 的答案是五步预处理 + 最长递增子序列（LIS）。 这是前端算法面试的天花板题，也是本章的全部内容。

18.1 patchElement：属性打补丁

先热身。新旧 vnode 同类型时，不走推倒重来，而是「就地打补丁」：

function patchElement(oldVNode, vnode) {
  const el = (vnode.el = oldVNode.el);   // ⭐ 复用旧 DOM！这一行就是「不推倒重来」

  // ① 属性 diff：新旧 props 的键取并集，逐键比较
  const oldProps = oldVNode.props || {};
  const newProps = vnode.props || {};
  for (const key in newProps) {
    if (newProps[key] !== oldProps[key]) {          // 变了的才动
      hostPatchProp(el, key, oldProps[key], newProps[key]);
    }
  }
  for (const key in oldProps) {
    if (!(key in newProps)) {                        // 新里没有 → 删属性
      hostPatchProp(el, key, oldProps[key], null);
    }
  }

  // ② children diff：本章主角
  patchChildren(oldVNode, vnode, el);
}

属性 diff 的哲学和第 6 章依赖清理、第 15 章 patchStyle 完全同构：新来的应用，过期的清除，没变的不碰。

18.2 patchChildren 的分诊台

function patchChildren(oldVNode, vnode, el) {
  const oldCh = oldVNode.children, newCh = vnode.children;

  if (typeof newCh === "string") {
    // 新是文本：旧的不管是啥，整体覆盖（数组的话先逐个卸载）
    if (Array.isArray(oldCh)) oldCh.forEach(c => unmount(c));
    if (oldCh !== newCh) hostSetElementText(el, newCh);
  } else if (Array.isArray(newCh)) {
    if (Array.isArray(oldCh)) {
      patchKeyedChildren(oldCh, newCh, el);   // ⭐ 双数组：进入主战场
    } else {
      hostSetElementText(el, "");             // 旧文本 → 清空再挂新数组
      newCh.forEach(c => patch(null, c, el, null));
    }
  } else {
    // 新 children 为空：旧的清光
    if (Array.isArray(oldCh)) oldCh.forEach(c => unmount(c));
    else hostSetElementText(el, "");
  }
}

18.3 主战场：五步预处理 + LIS

patchKeyedChildren 是 Vue3 渲染器最复杂的 100 行。先别被吓到 —— 用一个生活场景把整套算法「翻译」一遍：

🎨 一分钟比喻：全班换座位
期末换座位，旧座位表要换成新座位表，你是班主任。最省力的策略：
① 先扫两头：第一排和最后一排很多人压根没换 —— 这些人「白捡」，打声招呼（patch 内容）就行，不用搬桌子；
② 剩下的中间区才真的乱了。给每人发一本「通讯录」（key 映射表），按名字 O(1) 查到新座位；
③ 通讯录里查无此人 = 转学生（新挂载）；旧表里有、新表里没有 = 转学了（卸载）；
④ 灵魂问题是「最少搬几个人？」—— 找出相对顺序没变的最长一队人，让他们坐住不动（LIS），其他人对着这队人挪。 「最长递增子序列」的全部直觉就一句话：不动的人越多，要搬的人越少。

而且这套算法很懂「见好就收」：头尾预处理之后，大量日常场景根本走不到 LIS —— 尾部追加、头部插入、纯删除，都在中间几步直接收工。LIS 只属于真正的「乱序混合战」。 下面先把每一格推演清楚，再到 18.4 的分步实验室亲手走一遍。

第一步：从头同步（sync from start）

ab c← 停！ 前面 key 相同的逐个 patch（打补丁不移动），到 c 停下

第二步：从尾同步（sync from end）

头尾两次「白捡」之后，剩下中间一段真正乱序的。若一方已耗尽，直接批量挂载/卸载收工（第三步，略）。

第四步：乱序段 —— key 映射表

// 给新节点建索引表：key → 在新列表中的下标
const keyToNewIndex = new Map();
for (let j = s2; j <= e2; j++) keyToNewIndex.set(newCh[j].key, j);

// 遍历旧乱序段：在表里找得到 → 复用；找不到 → 卸载
const toBePatched = e2 - s2 + 1;
const newIndexToOldIndex = new Array(toBePatched).fill(-1);  // ⭐ 核心数组
let moved = false, maxNewIndexSoFar = 0;

for (let i = s1; i <= e1; i++) {
  const newIndex = keyToNewIndex.get(oldCh[i].key);
  if (newIndex === undefined) {
    unmount(oldCh[i]);                        // 新列表没有我 → 卸载
  } else {
    newIndexToOldIndex[newIndex - s2] = i;    // 记下「新位置的旧主人」
    if (newIndex < maxNewIndexSoFar) moved = true;  // 顺序倒退了 → 需要移动
    else maxNewIndexSoFar = newIndex;
  }
}

newIndexToOldIndex 是整个算法的枢纽：下标 = 新位置，值 = 这个位置的旧节点是谁。 -1 表示这个新位置没有旧主人（要新建）。官方实现里它初始化为 0、存的值是「旧下标 + 1」—— 这样 0 就代表「新挂载」，教学版改用 -1 更直观，思想一致。

第五步：LIS —— 能不动就不动

如果 moved = true，问题变成：newIndexToOldIndex 里哪些节点保持相对顺序不变（不用动），剩下的才需要移动？ 这正是最长递增子序列（Longest Increasing Subsequence）的定义。LIS 里的节点原地不动，其余的用 insert 挪到位 —— 数学保证这就是最少移动方案。 

// 目标：返回最长递增子序列的「下标集合」（注意：是下标，不是值！）
function getSequence(arr) {
  const result = [0];            // result 存下标；对应值单调递增
  for (let i = 1; i < arr.length; i++) {
    if (arr[i] === -1) continue;              // -1 = 新挂载位置，跳过
    const last = arr[result[result.length - 1]];
    if (arr[i] > last) {
      result.push(i);                         // 比尾巴大 → 直接接上（LIS 变长）
    } else {
      // 二分找到 result 中第一个 ≥ arr[i] 的位置，替换之（给后面留「更低的台阶」）
      let lo = 0, hi = result.length - 1;
      while (lo < hi) {
        const mid = (lo + hi) >> 1;
        if (arr[result[mid]] < arr[i]) lo = mid + 1;
        else hi = mid;
      }
      result[lo] = i;
    }
  }
  return result;   // 这些下标对应的节点 → 原地不动
}

// 最后一步：从后往前遍历新列表（从后往前是为了算 insert 的 anchor）
let lisIndex = result.length - 1;
for (let j = toBePatched - 1; j >= 0; j--) {
  const oldIndex = newIndexToOldIndex[j];
  if (oldIndex === -1) {
    patch(null, newCh[s2 + j], el, anchor);    // 没有旧主人 → 新建
  } else {
    if (lisIndex < 0 || j !== result[lisIndex]) {
      hostInsert(newCh[s2 + j].el, el, anchor); // 不在 LIS 里 → 挪我
    } else {
      lisIndex--;                                // 在 LIS 里 → 纯 patch，不挪
    }
  }
  anchor = newCh[s2 + j].el;                    // 锚点随行
}
💡 LIS 为什么「从后往前」遍历？
insert(el, container, anchor) 的语义是「插到 anchor 前面」。从后往前处理时，anchor 永远指向「已经处理好的下一个位置」， 每个节点挪到 anchor 前就自动落位。从前往后的话 anchor 语义会打架。这个方向选择是理解 diff 实现的钥匙之一。
⚠️ 官方的 getSequence 比上面多一步「回溯」
贪心 + 二分的裸 result 有个隐患：中间发生过「替换」的话，result 里留下的下标可能不再互为递增 （替换是为了给后面留更低的台阶，但被换下来的下标没清出去）。官方实现额外维护一个 前驱数组 p：每次替换时记下「我踩在谁头上」，最后从尾部沿 p 回溯出真正合法的 LIS。 完整版见 renderer.ts 的 getSequence —— 面试手撕 LIS 时，写出 p 数组回溯就是超越 90% 候选人的细节。

逐格推演：旧 [a,b,c,d,e] → 新 [a,d,c,b,f,e]

固定一个「五脏俱全」的例子 —— 有头同步、有尾同步、有乱序、有新增，跟着表走一遍：

追问一：第 18.3 节最后为什么要「从后往前」遍历 toBePatched？

追问二：Vue2 的双端 diff 挺好，为什么 Vue3 换了？

方案思路最坏复杂度致命弱点
Vue2 双端头尾四指针互夹O(n)乱序区仍要逐个找、逐个挪；四指针之外的节点要走遍历查找
Vue3 预处理 + LIS先头尾同步，中间乱序区求「最多不动」O(n log n)依赖 key（没 key 走退化路径，见下）

决定性因素其实是编译器的配合：Vue3 模板编译后每个动态子节点带 patchFlag + key 信息， 中间乱序段可以直接用 key 映射表 O(1) 定位，再配 LIS 把「必须动」的压到最少 —— 双端 diff 在纯运行时设计里很优，但它浪费了编译期已知的情报。 这是「编译反哺运行时」在 diff 算法选择上的直接体现（第 30 章）。

⚠️ 无 key 的退化路径
v-for 不给 key 时走 patchUnkeyedChildren：纯粹按下标对齐， 下标相同就地 patch，多退少补 —— 没有移动判断，但内容错位（第 25.5 节源码）。 「为什么不能用 index 当 key」的本质：index 对齐在删除中间项时会把「内容」错配给 「身份」，状态（输入框内容、勾选）全部串位 —— 不是 diff 慢，是身份错了。

function patch(oldVNode, vnode, container, anchor) {
  if (oldVNode === vnode) return;
  if (oldVNode && oldVNode.type !== vnode.type) { unmount(oldVNode); oldVNode = null; }

19.2 mountComponent：三步走

function mountComponent(oldVNode, vnode, container, anchor) {
  // ① 创建组件实例：一个装所有运行时状态的「档案袋」
  const instance = (vnode.component = {
    vnode, props: {}, slots: null, subTree: null,   // subTree = 组件渲染出的元素树
    isMounted: false, next: null,                    // next = 待更新的新 vnode
  });

  // ② 初始化 props + 调用 setup（你写的组合式 API 在这里执行！）
  instance.props = vnode.props;
  const setupResult = vnode.type.setup?.(instance.props);
  if (typeof setupResult === "function") {
    instance.render = setupResult;      // setup 返回函数 → 就是 render
  } else {
    instance.setupState = proxyRefs(setupResult);  // 返回对象 → 第 7 章的自动脱 .value！
    instance.render = vnode.type.render;
  }

  // ③ 建立响应式渲染：⭐⭐ 全书最重要的一次 effect 包装 ⭐⭐
  setupRenderEffect(instance, container, anchor);
}

function setupRenderEffect(instance, container, anchor) {
  instance.update = effect(function componentUpdateFn() {
    if (!instance.isMounted) {
      // 首挂载：跑 render 拿到元素子树，递归挂载
      const subTree = instance.render();
      patch(null, subTree, container, anchor);
      instance.subTree = subTree;
      instance.isMounted = true;
    } else {
      // 更新：重跑 render 拿新子树，与旧 subTree 做 patch（第 18 章 diff 上场）
      const nextTree = instance.render();
      const prevTree = instance.subTree;
      patch(prevTree, nextTree, container, anchor);
      instance.subTree = nextTree;
    }
  });
}

数据变化 trigger → update（effect）重跑 → 新旧 subTree diff → 最小化更新 DOM

19.3 父组件更新，子组件怎么办？

父组件重渲染时，子组件 vnode 也会重建。但子组件不一定要重渲染 —— Vue 的组件更新是有「资格审查」的（shouldUpdateComponent）： 

function updateComponent(oldVNode, newVNode) {
  newVNode.component = oldVNode.component;   // ⭐ 复用整个实例（组件的「身份」不变）

  if (shouldUpdateComponent(oldVNode, newVNode)) {
    newVNode.component.next = newVNode;      // 待更新 vnode 挂到 next 上（异步排队）
    newVNode.component.update();             // 手动触发一次渲染 effect
  } else {
    // props 没变（浅比较 + hasChanged）：直接吃掉新 vnode，跳过重渲染
    newVNode.component.vnode = newVNode;
  }
}

// shouldUpdateComponent 的粗略逻辑：
//   - 新旧 props 里有任何一个键的值变了（Object.is 比较）→ true
//   - 子组件还有 slots/emits 依赖 → 保守起见 true
🧠 这就是「组件级缓存」的第一道防线
props 没变就跳过整棵子树的重渲染 —— 比 memo 手动优化更早、更自动。 配合第 30 章编译器的 patchFlag（精确知道哪些 props 可能变），这道闸门效率极高。 Vue 里 v-memo、React 里 memo + 浅比较，本质都在做同一件事：身份没变就别动。

19.4 生命周期：注册即挂榜

onMounted 等钩子的原理朴素得可爱：往当前组件实例的钩子数组里 push 回调，时机到了挨个调用。 关键又是那个熟悉的全局指针模式： 

let currentInstance = null;          // 第三次见到这个模式了！（activeSub、activeEffectScope）

  } else {
    warn("onMounted 只能在 setup 中调用");

function setupComponent(instance) {
  const last = currentInstance;      // 暂存（第 10 章的老配方）
  currentInstance = instance;
  try {
    instance.render = instance.type.setup(instance.props) || ...;
  } finally {
    currentInstance = last;          // 还原 —— 所以异步回调里调 onMounted 会丢
  }

function mountComponent(instance) {
  /* ...挂载完成后... */
  instance.mounted.forEach(fn => fn());   // 挨个执行（源码里叫 invokeArrayFns）

⚠️ 为什么 async setup 里不能裸调 onMounted？
setup 一旦 await，执行权就交出去了 —— 后续代码运行时 currentInstance 早已还原成 null，钩子注册失败。 所以 Vue 提供 onActivated/onDeactivated 之外还有 Suspense 的特殊处理， 而普通项目里的忠告是：生命周期注册必须在 setup 的同步阶段完成。原理现在你懂了：不是规则霸道，是指针回家了。

19.5 更新调度：effect 的 scheduler 第二春

真实 Vue 里 instance.update 的 scheduler 不是立刻执行，而是 queueJob —— 推进一个去重队列，微任务里统一 flush。 为什么？一个事件里改 3 个数据，trigger 3 次 update，但 patch 只需要最后一次： 

const queue = new Set();          // Set 天然去重：同一组件的 update 只留一份
let isFlushing = false;

function queueJob(job) {
  queue.add(job);                 // 只排队，不执行
  if (!isFlushing) {
    isFlushing = true;
    Promise.resolve().then(flushJobs);   // 微任务：等本轮同步代码全跑完
  }
}

function flushJobs() {
  queue.forEach(job => job());    // 此刻数据已是最终态，每个组件只 patch 一次
  queue.clear(); isFlushing = false;
}

第 5 章 scheduler 的种子（「数据变了不直接跑，先问调度器」）至此完全开花结果。不过这只是骨架 —— 第 23 章会把它补齐到生产级：按 id 排序保证父先子后、pre/post 两支插队队伍，以及 nextTick 的真身。

19.6 this 的真身：PublicInstanceProxyHandlers 的查找顺序

你在模板里写 {{ count }}、在 options API 里写 this.count —— 这些访问背后是一层「渲染代理」， 它按固定顺序在实例的各个口袋里找值： 

const PublicInstanceProxyHandlers = {
  get({ _: instance }, key) {
    const { ctx, setupState, data, props, accessCache } = instance;
    // ⭐ accessCache：第一次查找的结果缓存成编号，之后 O(1) 直达
    let normalizedProps;
    if (key[0] !== '$') {
      // ⭐ accessCache：首次查找的结果缓存成编号（空间换时间）。
      //    官方 AccessTypes：SETUP=1 / DATA=2 / PROPS=3 / CONTEXT=4；教学版压缩成 0~3，顺序一致
      const n = accessCache[key];
      if (n !== undefined) {
        if (n === 0) return setupState[key];
        else if (n === 1) return data[key];
        else if (n === 3) return ctx[key];
        else if (n === 2) return props[key];
      } else if (hasSetupBinding(setupState, key)) {
        accessCache[key] = 0;
        return setupState[key];                  // ① setup 返回的绑定（script setup 最快路径）
      } else if (data !== EMPTY_OBJ && hasOwn(data, key)) {
        accessCache[key] = 1;
        return data[key];                        // ② data
      } else if (
        (normalizedProps = instance.propsOptions[0]) &&
        hasOwn(normalizedProps, key)
      ) {
        accessCache[key] = 2;
        return props[key];                       // ③ props
      } else if (hasOwn(ctx, key)) {
        accessCache[key] = 3;
        return ctx[key];                         // ④ ctx：方法 / computed / $ 开头的内置属性
      }
    }
  },
  has({ _: { data, setupState, ctx, propsOptions, appContext } }, key) {
    let normalizedProps;
    return (
      key !== '!' && !(
        (key[0] !== '$' &&
          (hasSetupBinding(setupState, key) ||
            (data !== EMPTY_OBJ && hasOwn(data, key)) ||
            (normalizedProps = propsOptions[0]) && hasOwn(normalizedProps, key) ||
            hasOwn(ctx, key))) ||
        hasOwn(appContext.config.globalProperties, key)
      )
    );
  },
  set({ _: instance }, key, value) {
    const { data, setupState, ctx } = instance;
    if (hasSetupBinding(setupState, key)) {
      setupState[key] = value;                   // 改 setup 绑定 → 触发响应式（第 3 章）
      return true;
    } else if (setupState.__isScriptSetup && hasOwn(setupState, key)) {
      warn('Cannot mutate <script setup> binding from outside.');
      return false;                              // ⭐ script setup 的绑定从外部只读
    } else if (data !== EMPTY_OBJ && hasOwn(data, key)) {
      data[key] = value;
      return true;
    } else if (hasOwn(instance.props, key)) {
      warn('Attempting to mutate prop "' + key + '". Props are readonly.');
      return false;                              // props 只读（第 22 章 shallowReactive）
    }
    ctx[key] = value;
    return true;
  },

① <script setup> 的变量最快（31.5 的编译期结论在运行时的对应物）：它排在查找链第一位；
② props 只读不是约定是代码：set 钩子对 props 赋值直接拦截 + 警告；

19.7 shouldUpdateComponent 的完整判定表

19.3 给的是思想版，官方真实判定是一条优先级短路链 —— 从最快的检查往下走：

function shouldUpdateComponent(prevVNode, nextVNode, optimized) {
  const { props: prevProps, children: prevChildren } = prevVNode;
  const { patchFlag, props: nextProps, children: nextChildren } = nextVNode;

  if (patchFlag > 0) {                                  // ⭐ 编译产物走这里
    if (patchFlag & PatchFlags.DYNAMIC_SLOTS) return true;   // 动态插槽 → 必更
    if (patchFlag & PatchFlags.PROPS) {
      for (const key of nextVNode.dynamicProps!) {          // ⭐ 只看编译期登记的动态键！
        if (nextProps[key] !== prevProps[key]) return true;
      }
    }
  } else if (!optimized) {                              // 手写渲染函数：全量判断
    if (prevChildren || nextChildren) {
      if (!nextChildren) return true;                   // 插槽从有到无
      if (!prevChildren) return true;                   // 插槽从无到有
    }
    if (prevProps === nextProps) return false;          // 引用相等 → 短路跳过
    if (!prevProps) return !!nextProps;
    if (!nextProps) return true;
    return hasPropsChanged(prevProps, nextProps);       // 逐键 Object.is（19.3）
  }
  return false;                                         // ⭐ 默认不更新
}
情形判定
编译产物 + dynamicProps 全没变false —— 子组件连 render 都不执行（日常最常见快路径）

第 20 章 · 渲染器篇

Block Tree 的运行时消费

第 30 章讲了编译器怎么生成 dynamicChildren（跳过清单）， 本章补上另一半：运行时怎么消费这份清单。openBlock / createBlock 这两个你天天在编译产物里看到的函数， 合起来不过 30 行 —— 但它们是「靶向更新」的发动机。

20.1 Block 栈：30 行的收集器

// ⭐ 全局 Block 栈：支持嵌套（v-if/v-for 各自开新 Block）
let currentBlock: (VNode[] | null)[] = [];

export function openBlock(disableTracking = false) {
  currentBlock.push(disableTracking ? null : []);
}

export function createElementBlock(...) {
  return setupBlock(createBaseVNode(...));
}

function setupBlock(vnode) {
  vnode.dynamicChildren = isBlockTreeEnabled
    ? currentBlock.pop() || []    // ⭐ 弹出栈顶 = 我这个 Block 收集到的全部动态节点
    : [];
  // 自己作为子 Block，还要被父 Block 收集
  if (currentBlock.length) currentBlock[currentBlock.length - 1].push(vnode);
  return vnode;
}

// 普通动态子节点（createVNode 内部）：
if (currentBlock && vnode.patchFlag > 0) {
  currentBlock[currentBlock.length - 1].push(vnode);   // 扁平收集！不按层级
}
🧠 关键设计：扁平收集，不按层级
dynamicChildren 是一个扁平数组，深度为 10 的嵌套里某个动态文本，和顶层动态属性在同一个清单里。 传统 diff 要「逐层递归找出变化」，Block 只需「遍历清单逐个 patch」—— 静态层级被整个跳过，这就是编译期信息换来的运行时豁免。

20.2 patchBlockChildren：对比清单而非对比树

const patchBlockChildren = (oldChildren, newChildren, fallbackContainer) => {
  // ⭐ 只 diff 两个动态清单（长度天然一致：同一模板编译产物）
  for (let i = 0; i < newChildren.length; i++) {
    const oldVNode = oldChildren[i];
    const newVNode = newChildren[i];
    patch(oldVNode, newVNode, fallbackContainer, null, ...);  // 逐个走 patch，但只在清单内
  }
  // props/指令仍按 patchFlag 快慢车道处理（15.7）
};
旧 diff：整棵树递归 → 100 个节点全部走一遍 isSameVNodeType
Block diff：查清单 → 只处理 dynamicChildren 里的 3 个动态节点 → 97 个静态节点零成本
更新成本从 O(全部节点) 降到 O(动态节点) —— 页面越静态，收益越大

20.3 边界：v-if / v-for 为什么必须「开新 Block」

场景为什么打断收集
v-if 分支切换子树结构可能完全不同 —— 旧清单作废，新分支自己开 Block 重新收集
v-for 列表长度可变 → 每个 item 开新 Block，diff 只在清单间进行（32.5）
h() 手写渲染没有编译信息，isBlockTreeEnabled 下的收集照跑，但无 patchFlag 意义有限
手动 openBlock(true)disableTracking：确定无动态内容的 Block 不收集，进一步省内存
⚠️ 手写渲染函数的「裸奔」
直接用 h() 写渲染函数时没有 openBlock/createBlock 包裹， 更新走完整 patch —— 不是 bug，是「没有编译情报就得付全价」。 这也解释了为什么官方坚持「模板优先」：同样的组件，模板编译产物天然带 Block + patchFlag， 手写 JSX/渲染函数两样都没有。React 选择优化运行时来补偿，Vue 选择扩大编译期职权。
自测一下：Block Tree 运行时

1. dynamicChildren 里的节点还需要递归 diff 吗？
2. openBlock 为什么设计成栈？

答案：
1. 清单内的节点仍然 patch（属性/文本可能变），但不再整树递归找它们 —— 位置由编译期确定，清单顺序在两次渲染间保持一致（结构变化会开新 Block）。
2. 因为 Block 可以嵌套：v-if/v-for 子树各开新 Block，先收集完的子 Block 先弹出并登记到父 Block —— 栈恰好表达「子 Block 完成时回归父 Block 继续收集」。
